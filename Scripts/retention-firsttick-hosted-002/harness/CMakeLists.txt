cmake_minimum_required(VERSION 3.16)
project(RetentionForegroundQualification LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
  message(FATAL_ERROR "This workload requires the reviewed uniform optimized Core build")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../source" core EXCLUDE_FROM_ALL)
add_executable(RetentionForegroundProbe probe.cpp)
target_link_libraries(RetentionForegroundProbe PRIVATE LatticeCore Threads::Threads ${CMAKE_DL_LIBS})
if(APPLE)
  target_link_options(RetentionForegroundProbe PRIVATE "-Wl,-map,${CMAKE_BINARY_DIR}/probe.map")
else()
  target_link_options(RetentionForegroundProbe PRIVATE "-Wl,-Map,${CMAKE_BINARY_DIR}/probe.map")
endif()
